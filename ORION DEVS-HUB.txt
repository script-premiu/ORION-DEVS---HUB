-- ==========================================
-- CONFIGURACIÓN INICIAL DE ORION DEVS
-- ==========================================

-- 1. Cargamos la librería de la interfaz (Rayfield UI)
local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

-- 2. Creamos la ventana principal con tu nombre y color azul
local Window = Rayfield:CreateWindow({
   Name = "ORION DEVS", -- El nombre de tu Hub
   LoadingTitle = "Cargando ORION DEVS...",
   LoadingSubtitle = "by TuNombre",
   Icon = "https://imgur.com/a/dM51ILV",
   ConfigurationSaving = {
      Enabled = true,
      FolderName = "OrionDevsConfig", -- Se guardará la config del usuario aquí
      FileName = "Config"
   },
   -- Aquí cambiamos el color a AZUL (Formato RGB)
   Theme = {
       Default = Color3.fromRGB(0, 120, 255), -- Azul brillante
       Accent = Color3.fromRGB(0, 80, 200),   -- Azul más oscuro para detalles
       Outline = Color3.fromRGB(0, 150, 255), -- Borde azul claro
       Text = Color3.fromRGB(255, 255, 255),  -- Texto blanco
       Background = Color3.fromRGB(15, 15, 25), -- Fondo oscuro azulado
       Button = Color3.fromRGB(0, 100, 220)   -- Color de los botones
   },
   KeySystem = false -- Ponlo en true si quieres ponerle contraseña
})

-- ==========================================
-- FUNCIÓN MÁGICA PARA CARGAR SCRIPTS
-- ==========================================

-- Esta función recibe una URL, la descarga y la ejecuta
local function CargarScriptExterno(url)
    local success, errorMsg = pcall(function()
        loadstring(game:HttpGet(url))()
    end)
    
    if success then
        Rayfield:Notify({
            Title = "ORION DEVS",
            Content = "Script cargado exitosamente.",
            Duration = 3,
            Image = 4483362458 -- Un icono de check
        })
    else
        Rayfield:Notify({
            Title = "Error en ORION DEVS",
            Content = "No se pudo cargar el script. Revisa la URL.",
            Duration = 5,
            Image = 4483362458
        })
    end
end

-- ==========================================
-- CREACIÓN DE PESTAÑAS (TABS)
-- ==========================================

-- Pestaña Principal (Aquí irán tus scripts)
local MainTab = Window:CreateTab("Scripts Principales", 4483362458) 

-- Pestaña de Configuración (Opcional)
local SettingsTab = Window:CreateTab("Ajustes", 4483362458)

-- ==========================================
-- AGREGANDO LOS BOTONES DE TUS SCRIPTS
-- ==========================================

-- Botón 1: Shader
MainTab:CreateButton({
   Name = "shaders",
   Callback = function()
       -- REEMPLAZA ESTA URL CON LA DE TU SCRIPT DE AUTOFARM
       CargarScriptExterno("https://raw.githubusercontent.com/script-premiu/Shaders/refs/heads/main/1Shaders.txt")
   end,
})

-- Botón 2: TP jugadores
MainTab:CreateButton({
   Name = "tp jugadores",
   Callback = function()
       -- REEMPLAZA ESTA URL CON LA DE TU SCRIPT DE VELOCIDAD
       CargarScriptExterno("https://raw.githubusercontent.com/script-premiu/Shaders/refs/heads/main/1TP-JUGADORES.txt")
   end,
})

-- Botón 3: CARFLY
MainTab:CreateButton({
   Name = "carfly",
   Callback = function()
       -- REEMPLAZA ESTA URL CON LA DE TU SCRIPT DE SALTO
       CargarScriptExterno("https://raw.githubusercontent.com/script-premiu/CARFLY-ROBO/refs/heads/main/1A%20CARFLY-ROBO.txt")
   end,
})

-- ==========================================
-- AGREGANDO UN SLIDER (EJEMPLO DE AJUSTE)
-- ==========================================

-- Si quieres un control deslizante para la velocidad
MainTab:CreateSlider({
   Name = "Ajustar Velocidad",
   Range = {16, 500}, -- De 16 a 500
   Increment = 1,
   Suffix = "Velocidad",
   CurrentValue = 16,
   Flag = "SliderVelocidad", -- Para guardar la config
   Callback = function(Value)
       -- Esto cambia la velocidad directamente sin necesidad de script externo
       game.Players.LocalPlayer.Character.Humanoid.WalkSpeed = Value
   end,
})

-- ==========================================
-- AJUSTES EXTRA
-- ==========================================

SettingsTab:CreateButton({
   Name = "Destruir Interfaz (Cerrar Hub)",
   Callback = function()
       Rayfield:Destroy()
   end,
})
