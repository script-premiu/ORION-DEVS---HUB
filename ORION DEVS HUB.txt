-- ==========================================
-- FUNCIÓN SEGURA PARA CARGAR SCRIPTS
-- ==========================================

local function CargarScriptExterno(url, nombre)
    local success, errorMsg = pcall(function()
        loadstring(game:HttpGet(url))()
    end)
    
    if success then
        print("[ORION DEVS] Cargado exitosamente: " .. tostring(nombre))
    else
        warn("[ORION DEVS] Error en " .. tostring(nombre) .. ": " .. tostring(errorMsg))
    end
end

-- SISTEMA DE ARRASTRE PARA LA VENTANA PRINCIPAL
-- ==========================================

local function HacerVentanaArrastrable(instanciaArrastrable, areaClic)
    local dragging, dragInput, dragStart, startPos
    areaClic = areaClic or instanciaArrastrable

    local function update(input)
        local delta = input.Position - dragStart
        instanciaArrastrable.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end

    areaClic.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = instanciaArrastrable.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    areaClic.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            update(input)
        end
    end)
end

HacerVentanaArrastrable(MainFrame, Header)

-- ==========================================
-- SISTEMA INTELIGENTE PARA EL BOTÓN FLOTANTE
-- ==========================================

local function ConfigurarBotonFlotanteInteligente(btn)
    local dragging = false
    local dragInput, dragStart, startPos
    local clickStartPosition = Vector2.zero

    local function update(input)
        local delta = input.Position - dragStart
        btn.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end

    btn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            clickStartPosition = Vector2.new(input.Position.X, input.Position.Y)
            startPos = btn.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    -- Calcular la distancia que se movió el cursor desde que dio clic
                    local endPosition = Vector2.new(input.Position.X, input.Position.Y)
                    local distanceMoved = (endPosition - clickStartPosition).Magnitude
                    
                    -- Si se movió menos de 5 píxeles, cuenta como un Clic REAL para abrir la GUI
                    if distanceMoved < 5 then
                        MainFrame.Visible = true
                        OpenBtn.Visible = false
                    end
                end
            end)
        end
    end)

    btn.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            update(input)
        end
    end)
end

ConfigurarBotonFlotanteInteligente(OpenBtn)

-- LÓGICA DE MINIMIZAR Y TECLA DE ACCESO

local function ToggleUI()
    MainFrame.Visible = not MainFrame.Visible
    OpenBtn.Visible = not MainFrame.Visible
end

    OrionDevsHub:Destroy()
end)

UserInputService.InputBegan:Connect(function(input, gpe)
    if not gpe and input.KeyCode == Enum.KeyCode.RightControl then
        ToggleUI()
    end
end)
